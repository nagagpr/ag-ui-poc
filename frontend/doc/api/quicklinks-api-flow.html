Real text: quicklinks/sql/get_quick_links_queries.py. Expected shape:

WITH caller AS ( SELECT ... WHERE email = %s ), ...
SELECT ql.id, ql.url_label, ql.category, ql.url, ql.open_in_new_tab, ql.display_order,
       ql.quicklink_display_start_date, ql.quicklink_display_end_date,
       ql.program, ql.program_instance, ql.model_type, ql.role_type, ql.special_feature, ql.cpp_id,
       CASE WHEN ql.quicklink_display_end_date < CURRENT_DATE THEN 'Expired' ELSE ql.status END AS status,
       ql.created_at, ql.updated_at, ql.created_by, ql.updated_by,
       COUNT(*) OVER () AS total_count
FROM <quick links table> ql
WHERE ql.is_current = TRUE
  AND ( ... is_internal / participation rules ... )
  AND (%s::int[] IS NULL OR ql.id = ANY(%s))
ORDER BY ql.display_order
LIMIT %s OFFSET %s;

4. POST/v1/quick-links — create CONFIRMED

POST {{baseUrl}}/v1/quick-links
Authorization: Bearer <token>   Accept: application/json   Content-Type: application/json
Body: QuickLinkCreateRequest
  1. Body parsed into QuickLinkCreateRequest — Pydantic validators run (absolute HTTPS URL, label, dates, category fields).
  2. validate_accept_header + validate_media_type.
  3. validate_content_type(content_type, accept) — body must be JSON.
  4. validate_internal_access — external users → 403.
  5. quick_links_service.create_quick_link(request, user_email) → QuickLinksCreateService → create_quick_link_queries.py (CTEs validate CPP IDs / participation, then INSERT).
  6. ErrorResponse → its status (409 duplicate…); success → convert_model_to_response(result, accept, "QuickLink", 201).
  7. Exception → 500 "Error creating quick link".

5. PATCH/v1/quick-links/{id}?action=… — update or soft delete CONFIRMED

# UPDATE
PATCH /v1/quick-links/12?action=<UPDATE value>
{ "updated_at": "<from last GET>",
  "category": "...",
  "url_label": "New label" }
# DELETE (soft)
PATCH /v1/quick-links/12?action=<DELETE value>
{ "updated_at": "<from last GET>" }

VERIFY exact action values in ActionEnum (common/models.py).

  1. Accept / media type validation.
  2. _validate_action_param(action) — must be an ActionEnum value → else 400.
  3. _build_patch_request(action, body):
    • DELETE: only updated_at required; other fields ignored so validators can't block a delete.
    • UPDATE: updated_at and category required → QuickLinkPatchRequest(**body).
  4. validate_internal_access → external users 403.
  5. UPDATE → update_quick_link → 200 + QuickLink · DELETE → delete_quick_link → 200 {"message": "Quick link deleted successfully"}.
  6. Errors: 400, 401, 403, 404 not found, 409 version mismatch, 422 cannot edit expired/inactive, 500.
Optimistic locking: you send back the updated_at you received from GET. If someone changed the link meanwhile, the timestamps differ and you get 409 Conflict instead of overwriting their change.

GET_FAVORITES = """
    SELECT f.id, f.name, f.url, f.created_at, COUNT(*) OVER () AS total_count

    WHERE f.user_email = %s
    ORDER BY f.created_at DESC

Service — favorites/service/favorites_service.py 
logger = LoggingUtil.get_logger(__name__)

@lru_cache()
def get_favorites_service_singleton() -> "FavoritesService":
    db_name = settings.get_property("database", "portal_db_name")
    db_user = settings.get_property("database", "portal_db_user")
    return FavoritesService(get_manager().get_service(db_name=db_name, db_user=db_user))

class FavoritesService:
    def __init__(self, db):
        self.db = db

    def search_favorites(self, user_email, pagination) -> FavoriteListResponse | ErrorResponse:
        try:
            rows = self.db.execute_query(GET_FAVORITES,
                                         (user_email, pagination.limit, pagination.offset),
                                         fetch_one=False, use_named_tuples=True) or []
            total = rows[0].total_count if rows else 0
            items = [Favorite(id=r.id, name=r.name, url=r.url, created_at=r.created_at) for r in rows]
            return FavoriteListResponse(favorites=items, pagination=Pagination(
                total=total if pagination.offset == 0 else None,
                limit=pagination.limit, offset=pagination.offset, result_set_id=None,
                has_more=pagination.offset + len(items) < total))
        except Exception:
            logger.exception("Error searching favorites")
            return internal_server_error(ERROR_CODE_RETRIEVING_DATA, ERROR_MSG_RETRIEVING_DATA)

Router — favorites/router/favorites_router.py 
logger = LoggingUtil.get_logger(__name__)
router = APIRouter(tags=["Favorites"])

@router.get("", summary="Get favorites", response_model=FavoriteListResponse,
            responses={400: {"model": ErrorResponse}, 401: {"model": ErrorResponse},
                       500: {"model": ErrorResponse}})
async def get_favorites(
    request: Request,
    auth: Annotated[AuthService, Depends(get_auth_service)],
    user_context: Annotated[UserContext, Depends(get_user_context)],
    accept: Annotated[Optional[str], Header(alias="Accept")] = MediaType.APPLICATION_JSON,
    page_limit: Annotated[int, Query(ge=1, le=100, alias="limit")] = 100,
    page_offset: Annotated[int, Query(ge=0, alias="offset")] = 0,
    service: Annotated[FavoritesService, Depends(get_favorites_service_singleton)] = None,
) -> FavoriteListResponse | JSONResponse:
    accept = validate_accept_header(accept)
    error = validate_media_type(accept)
    if error:
        return convert_model_to_response(error, accept, "ErrorResponse", error.status)

    pagination = PaginationRequest(limit=page_limit, offset=page_offset, result_set_id=None)
    try:
        result = service.search_favorites(user_context.email, pagination)
        if isinstance(result, ErrorResponse):
            return convert_model_to_response(result, accept, "ErrorResponse", result.status)
        return convert_model_to_response(result, accept, "FavoriteListResponse")
    except Exception:
        logger.exception("Unhandled exception in get_favorites")
        err = internal_server_error(ERROR_CODE_RETRIEVING_DATA, ERROR_MSG_RETRIEVING_DATA)
        return convert_model_to_response(err, accept, "ErrorResponse", err.status)
